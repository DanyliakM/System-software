cmake_minimum_required(VERSION 3.16)
project(PhotoManager VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(NOT WIN32)
    message(FATAL_ERROR "PhotoManager requires Windows")
endif()

add_compile_definitions(
    UNICODE _UNICODE
    WIN32_LEAN_AND_MEAN NOMINMAX
    _WIN32_WINNT=0x0A00
    NTDDI_VERSION=0x0A000000
)

set(SOURCES
    main.cpp

    src/filesystem/file_ops.cpp
    src/filesystem/ntfs_ads.cpp
    src/filesystem/folder_watcher.cpp
    src/filesystem/registry_settings.cpp
    src/filesystem/operation_log.cpp

    src/hash/sha256.cpp
    src/hash/hasher.cpp

    src/exif/exif_parser.cpp

    src/catalog/catalog.cpp

    src/filter/search_engine.cpp

    src/rename/batch_renamer.cpp

    src/import/importer.cpp

    src/ui/thumbnail_cache.cpp
    src/ui/console_ui.cpp
)

add_executable(PhotoManager ${SOURCES})

target_include_directories(PhotoManager PRIVATE
    ${CMAKE_SOURCE_DIR}
    ${CMAKE_SOURCE_DIR}/src
)

target_link_libraries(PhotoManager PRIVATE
    shell32
    ole32
    oleaut32
    shlwapi
    user32
    advapi32
    gdi32
    comctl32
)

set_target_properties(PhotoManager PROPERTIES
    WIN32_EXECUTABLE FALSE
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin
)

message(STATUS "PhotoManager ${PROJECT_VERSION} -- Windows-only build")
